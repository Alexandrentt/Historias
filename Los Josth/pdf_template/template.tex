% Comando para epígrafe de capítulo
% Uso en markdown: 
% > Texto del epígrafe
% > — Autor, *Obra*
% Pandoc lo convierte a \begin{quote}...\end{quote}
% Este comando formatea un epígrafe elegante:
\newcommand{\capituloepigrafe}[2]{%
    \begin{epigraph}
    #1\\
    \vspace{0.5em}
    {\normalsize--- #2}
    \end{epigraph}
    \vspace{1em}
}

% Comando para insertar ilustración
% Se usa con el formato [ilustracion:descripcion] en markdown
% El script de compilación lo convierte a:
% \ilustracion{nombre_archivo}{descripcion}{numero}
\newcommand{\ilustracion}[3]{%
    \begin{figure}[H]
    \centering
    \includegraphics[width=0.9\textwidth]{#1}
    \caption{\textcolor{ilustracioncolor}{\small #2}}
    \label{fig:#3}
    \end{figure}
    \vspace{0.5em}
}

% Comando para placeholder de ilustración (cuando aún no existe la imagen)
\newcommand{\ilustracionplaceholder}[3]{%
    \begin{figure}[H]
    \centering
    \fbox{\parbox{0.8\textwidth}{%
        \centering
        \vspace{3cm}
        {\large\itshape [Ilustración #3]}\\[1em]
        {\small #2}
        \vspace{3cm}
    }}
    \caption{\textcolor{ilustracioncolor}{\small #2}}
    \label{fig:#3}
    \end{figure}
    \vspace{0.5em}
}
